using namespace std;

// 1. Enums & Entities
enum class BookingStatus { CREATED, HELD, CONFIRMED, CHECKED_IN, CHECKED_OUT, CANCELLED };

struct Booking {
    string id;
    string roomTypeId;
    long long checkInDate;
    long long checkOutDate;
    BookingStatus status;
};

// 2. Centralized State Handler
class BookingStateHandler {
public:
    static bool canTransition(BookingStatus current, BookingStatus nextStatus) {
        if (current == BookingStatus::CREATED && nextStatus == BookingStatus::HELD) return true;
        if (current == BookingStatus::HELD && nextStatus == BookingStatus::CONFIRMED) return true;
        if (current == BookingStatus::HELD && nextStatus == BookingStatus::CANCELLED) return true;
        if (current == BookingStatus::CONFIRMED && nextStatus == BookingStatus::CHECKED_IN) return true;
        if (current == BookingStatus::CONFIRMED && nextStatus == BookingStatus::CANCELLED) return true;
        if (current == BookingStatus::CHECKED_IN && nextStatus == BookingStatus::CHECKED_OUT) return true;
        return false;
    }

    static void transition(Booking& booking, BookingStatus nextStatus) {
        if (!canTransition(booking.status, nextStatus)) {
            throw runtime_error("Invalid booking state transition.");
        }
        booking.status = nextStatus;
    }
};

// 3. Service Layer
class InventoryService {
public:
    int getTotalRooms(string roomTypeId) { return 10; }
    
    // In a real system, queries DB: SELECT COUNT(*) WHERE status IN (HELD, CONFIRMED) AND dates overlap
    int getBookedCount(string roomTypeId, long long checkIn, long long checkOut) {
        return 8; // Stubbed: 8 rooms already booked/held
    }

    bool checkAvailability(string roomTypeId, long long checkIn, long long checkOut, int qty) {
        int available = getTotalRooms(roomTypeId) - getBookedCount(roomTypeId, checkIn, checkOut);
        return available >= qty;
    }
};

class BookingService {
    InventoryService inventory;
public:
    Booking createBooking(string roomTypeId, long long checkIn, long long checkOut) {
        // Validation Phase
        if (!inventory.checkAvailability(roomTypeId, checkIn, checkOut, 1)) {
            throw runtime_error("Not enough rooms available.");
        }

        // Creation Phase
        Booking b = {"BK123", roomTypeId, checkIn, checkOut, BookingStatus::CREATED};
        cout << "Booking CREATED. Please proceed to payment." << endl;
        return b;
    }

    void initiatePayment(Booking& booking) {
        // Double check inventory right before holding
        if (!inventory.checkAvailability(booking.roomTypeId, booking.checkIn, booking.checkOut, 1)) {
            throw runtime_error("Rooms sold out while you were waiting!");
        }

        BookingStateHandler::transition(booking, BookingStatus::HELD);
        cout << "Booking HELD. Inventory locked for 15 mins. Redirecting to payment gateway..." << endl;
    }

    void handlePaymentCallback(Booking& booking, bool paymentSuccess) {
        if (paymentSuccess) {
            BookingStateHandler::transition(booking, BookingStatus::CONFIRMED);
            cout << "Payment SUCCESS. Booking CONFIRMED." << endl;
        } else {
            BookingStateHandler::transition(booking, BookingStatus::CANCELLED);
            cout << "Payment FAILED. Booking CANCELLED. Inventory released." << endl;
        }
    }
};

// Usage
int main() {
    BookingService bookingSvc;
    
    try {
        // 1. Create
        Booking myBooking = bookingSvc.createBooking("RT_DELUXE", 1700000, 1700900);
        
        // 2. Pay
        bookingSvc.initiatePayment(myBooking);
        
        // 3. Callback
        bookingSvc.handlePaymentCallback(myBooking, true);
        
    } catch(const exception& e) {
        cout << "Error: " << e.what() << endl;
    }

    return 0;
}
