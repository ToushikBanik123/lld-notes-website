#include <string>
#include <unordered_map>
#include <vector>
#include <queue>
#include <thread>
#include <mutex>
#include <cstdlib>

using namespace std;

// 1. Core Entities
struct Player {
    string name;
    int position = 0;
};

class Dice {
public:
    static int roll() {
        return (rand() % 6) + 1;
    }
};

class Board {
public:
    int size;
    unordered_map<int, int> snakes;
    unordered_map<int, int> ladders;

    Board(int s = 100) : size(s) {}

    int getFinalPosition(int currentPos) {
        if (snakes.count(currentPos)) {
            cout << "🐍 Bitten by snake at " << currentPos << " -> " << snakes[currentPos] << endl;
            return snakes[currentPos];
        }
        if (ladders.count(currentPos)) {
            cout << "🪜 Climbed ladder at " << currentPos << " -> " << ladders[currentPos] << endl;
            return ladders[currentPos];
        }
        return currentPos;
    }
};

// 2. Game Session
class SnakeAndLadderGame {
private:
    Board board;
    queue<Player> players;
    bool gameWon = false;

public:
    SnakeAndLadderGame(Board b, vector<string> playerNames) : board(b) {
        for (const auto& name : playerNames) {
            players.push({name, 0});
        }
    }

    void play() {
        while (!gameWon) {
            Player current = players.front();
            players.pop();

            int roll = Dice::roll();
            cout << current.name << " rolled a " << roll << endl;

            int newPos = current.position + roll;

            if (newPos <= board.size) {
                newPos = board.getFinalPosition(newPos);
                current.position = newPos;
                cout << current.name << " moved to " << newPos << endl;

                if (newPos == board.size) {
                    cout << "🎉 " << current.name << " WINS THE GAME! 🎉" << endl;
                    gameWon = true;
                    return;
                }
            } else {
                cout << current.name << " needs exactly " << (board.size - current.position) << " to win. Staying put." << endl;
            }

            players.push(current);
        }
    }
};

// 3. Game Manager (Singleton)
class GameManager {
private:
    static GameManager* instance;
    static mutex mtx;
    vector<thread> activeGames;

    GameManager() {}

public:
    static GameManager* getInstance() {
        lock_guard<mutex> lock(mtx);
        if (!instance) instance = new GameManager();
        return instance;
    }

    void startNewGame(vector<string> players) {
        Board b(100);
        b.snakes[99] = 10;
        b.ladders[5] = 50;

        // Start game in a new thread
        activeGames.push_back(thread([b, players]() {
            SnakeAndLadderGame game(b, players);
            game.play();
        }));
    }

    void waitForAllGames() {
        for (auto& t : activeGames) {
            if (t.joinable()) t.join();
        }
    }
};

GameManager* GameManager::instance = nullptr;
mutex GameManager::mtx;

int main() {
    srand(time(0));
    GameManager* gm = GameManager::getInstance();
    
    gm->startNewGame({"Alice", "Bob"});
    gm->startNewGame({"Charlie", "Dave"});
    
    gm->waitForAllGames();
    return 0;
}
