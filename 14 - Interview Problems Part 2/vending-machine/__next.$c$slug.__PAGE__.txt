#include <string>
#include <map>
#include <memory>
#include <stdexcept>

using namespace std;

// 1. Entities
struct Product {
    string name;
    double price;
};

// Forward Declaration
class VendingMachine;

// 2. State Pattern Interface
class VendingMachineState {
public:
    virtual void selectProduct(VendingMachine* machine, string productName) = 0;
    virtual void insertMoney(VendingMachine* machine, double amount) = 0;
    virtual void dispenseProduct(VendingMachine* machine) = 0;
    virtual void cancelTransaction(VendingMachine* machine) = 0;
    virtual string getName() = 0;
    virtual ~VendingMachineState() = default;
};

// 3. Vending Machine Context
class VendingMachine {
public:
    unique_ptr<VendingMachineState> state;
    map<string, int> inventory;
    map<string, double> prices;
    
    string selectedProduct;
    double currentBalance = 0.0;

    VendingMachine(); // Initialization provided below

    void setState(unique_ptr<VendingMachineState> newState) {
        state = move(newState);
    }

    void selectProduct(string name) { state->selectProduct(this, name); }
    void insertMoney(double amount) { state->insertMoney(this, amount); }
    void dispenseProduct() { state->dispenseProduct(this); }
    void cancelTransaction() { state->cancelTransaction(this); }
};

// 4. Concrete States
class IdleState : public VendingMachineState {
public:
    void selectProduct(VendingMachine* machine, string productName) override;
    void insertMoney(VendingMachine* machine, double amount) override { throw runtime_error("Select a product first."); }
    void dispenseProduct(VendingMachine* machine) override { throw runtime_error("No product selected."); }
    void cancelTransaction(VendingMachine* machine) override { throw runtime_error("No transaction to cancel."); }
    string getName() override { return "IDLE"; }
};

class PaymentState : public VendingMachineState {
public:
    void selectProduct(VendingMachine* machine, string productName) override { throw runtime_error("Already processing payment."); }
    void insertMoney(VendingMachine* machine, double amount) override;
    void dispenseProduct(VendingMachine* machine) override { throw runtime_error("Payment not complete."); }
    void cancelTransaction(VendingMachine* machine) override;
    string getName() override { return "PAYMENT"; }
};

class DispenseState : public VendingMachineState {
public:
    void selectProduct(VendingMachine* machine, string productName) override { throw runtime_error("Please wait, dispensing."); }
    void insertMoney(VendingMachine* machine, double amount) override { throw runtime_error("Please wait, dispensing."); }
    void dispenseProduct(VendingMachine* machine) override;
    void cancelTransaction(VendingMachine* machine) override { throw runtime_error("Cannot cancel while dispensing."); }
    string getName() override { return "DISPENSE"; }
};

// State Implementation Details
void IdleState::selectProduct(VendingMachine* machine, string productName) {
    if (machine->inventory[productName] <= 0) {
        cout << "Out of stock!" << endl;
        return;
    }
    machine->selectedProduct = productName;
    machine->currentBalance = 0.0;
    cout << "Selected " << productName << ". Price: $" << machine->prices[productName] << endl;
    machine->setState(make_unique<PaymentState>());
}

void PaymentState::insertMoney(VendingMachine* machine, double amount) {
    machine->currentBalance += amount;
    double price = machine->prices[machine->selectedProduct];
    cout << "Inserted: $" << amount << ". Total: $" << machine->currentBalance << endl;
    
    if (machine->currentBalance >= price) {
        machine->setState(make_unique<DispenseState>());
        machine->dispenseProduct(); // Auto-dispense when fully paid
    } else {
        cout << "Remaining: $" << (price - machine->currentBalance) << endl;
    }
}

void PaymentState::cancelTransaction(VendingMachine* machine) {
    cout << "Transaction cancelled. Refunding $" << machine->currentBalance << endl;
    machine->currentBalance = 0;
    machine->selectedProduct = "";
    machine->setState(make_unique<IdleState>());
}

void DispenseState::dispenseProduct(VendingMachine* machine) {
    double price = machine->prices[machine->selectedProduct];
    double change = machine->currentBalance - price;
    
    machine->inventory[machine->selectedProduct]--;
    cout << "Dispensing " << machine->selectedProduct << "!" << endl;
    if (change > 0) cout << "Returning change: $" << change << endl;
    
    machine->currentBalance = 0;
    machine->selectedProduct = "";
    machine->setState(make_unique<IdleState>());
}

VendingMachine::VendingMachine() {
    state = make_unique<IdleState>();
    inventory["Coke"] = 5;
    prices["Coke"] = 1.50;
    inventory["Chips"] = 2;
    prices["Chips"] = 1.00;
}

// Usage
int main() {
    VendingMachine vm;
    vm.selectProduct("Coke");
    vm.insertMoney(1.00);
    vm.insertMoney(1.00); // Triggers dispense and returns change

    return 0;
}
