#include <string>
#include <unordered_map>
#include <mutex>
#include <stdexcept>
#include <memory>

using namespace std;

// 1. Entities
struct Wallet {
    string id;
    long long balanceMinor;
    bool isActive;
};

// 2. Lock Service
class LockService {
    unordered_map<string, shared_ptr<mutex>> locks;
    mutex globalMtx;
public:
    shared_ptr<mutex> getLock(const string& key) {
        lock_guard<mutex> g(globalMtx);
        if (!locks[key]) locks[key] = make_shared<mutex>();
        return locks[key];
    }
};

// 3. Wallet Service
class WalletService {
    unordered_map<string, Wallet> db;
    LockService lockService;

public:
    WalletService() {
        db["W1"] = {"W1", 5000, true}; // $50.00
        db["W2"] = {"W2", 1000, true}; // $10.00
    }

    void transfer(string fromId, string toId, long long amount) {
        if (amount <= 0) throw runtime_error("Invalid amount");
        if (fromId == toId) throw runtime_error("Self transfer not allowed");

        // Deadlock Prevention: Sort keys
        string lock1 = min(fromId, toId);
        string lock2 = max(fromId, toId);

        auto mtx1 = lockService.getLock(lock1);
        auto mtx2 = lockService.getLock(lock2);

        // Acquire strictly in sorted order
        scoped_lock locks(*mtx1, *mtx2); 

        // Re-fetch state under lock
        Wallet& wFrom = db[fromId];
        Wallet& wTo = db[toId];

        if (!wFrom.isActive || !wTo.isActive) throw runtime_error("Wallet inactive");
        if (wFrom.balanceMinor < amount) throw runtime_error("Insufficient funds");

        // Mutate
        wFrom.balanceMinor -= amount;
        wTo.balanceMinor += amount;

        cout << "Transfer Success! " << fromId << " -> " << toId << " : " << amount << endl;
    }
};

int main() {
    WalletService svc;
    svc.transfer("W1", "W2", 1500); // Transfers $15.00
    return 0;
}
