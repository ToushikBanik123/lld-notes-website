#include <string>
#include <vector>
#include <unordered_map>
#include <mutex>
#include <stdexcept>

using namespace std;

// 1. Enums & Entities
enum class SeatStatus { AVAILABLE, LOCKED, BOOKED };

struct Seat {
    string id;
    int price;
    SeatStatus status = SeatStatus::AVAILABLE;
};

class Show {
public:
    string showId;
    string movieId;
    unordered_map<string, Seat> seats;
    mutex showMutex; // Local lock for concurrent access

    Show(string id, string mId) : showId(id), movieId(mId) {}
};

// 2. Booking Service
class BookingService {
public:
    string reserveSeats(Show& show, vector<string> seatIds, string userId) {
        lock_guard<mutex> lock(show.showMutex); // Critical section

        // 1. Validation Phase
        for (const string& sId : seatIds) {
            if (show.seats[sId].status != SeatStatus::AVAILABLE) {
                throw runtime_error("Seat " + sId + " is no longer available.");
            }
        }

        // 2. Lock Phase (Temporary hold before payment)
        int totalAmount = 0;
        for (const string& sId : seatIds) {
            show.seats[sId].status = SeatStatus::LOCKED;
            totalAmount += show.seats[sId].price;
        }

        string bookingId = "BKG_" + to_string(rand());
        cout << "User " << userId << " reserved seats. Booking ID: " << bookingId << " Total: $" << totalAmount << endl;
        return bookingId;
    }

    void confirmBooking(Show& show, vector<string> seatIds, string bookingId) {
        lock_guard<mutex> lock(show.showMutex);
        for (const string& sId : seatIds) {
            show.seats[sId].status = SeatStatus::BOOKED;
        }
        cout << "Booking " << bookingId << " confirmed!" << endl;
    }
};

int main() {
    Show show("SHOW_1", "AVENGERS_1");
    show.seats["A1"] = {"A1", 15};
    show.seats["A2"] = {"A2", 15};
    
    BookingService bs;

    try {
        // User 1 tries to reserve A1 and A2
        string bId = bs.reserveSeats(show, {"A1", "A2"}, "USER_1");
        
        // Simulating Payment Success
        bs.confirmBooking(show, {"A1", "A2"}, bId);

        // User 2 tries to reserve A1
        bs.reserveSeats(show, {"A1"}, "USER_2"); // Will throw error
    } catch(const exception& e) {
        cout << "Error: " << e.what() << endl;
    }

    return 0;
}
