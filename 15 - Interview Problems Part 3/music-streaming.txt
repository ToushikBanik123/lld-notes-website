using namespace std;

// 1. Media Entities
struct Song {
    string id;
    long fileSize;
};

// 2. LRU Chunk Cache
class CacheService {
    int maxSize;
    list<string> lruKeys;
    unordered_map<string, pair<vector<uint8_t>, list<string>::iterator>> cache;

public:
    CacheService(int size) : maxSize(size) {}

    bool getChunk(const string& key, vector<uint8_t>& outChunk) {
        if (cache.find(key) != cache.end()) {
            lruKeys.erase(cache[key].second);
            lruKeys.push_front(key);
            cache[key].second = lruKeys.begin();
            outChunk = cache[key].first;
            return true;
        }
        return false;
    }

    void putChunk(const string& key, const vector<uint8_t>& chunk) {
        if (cache.size() >= maxSize) {
            string lru = lruKeys.back();
            lruKeys.pop_back();
            cache.erase(lru);
            cout << "[Cache] Evicted: " << lru << endl;
        }
        lruKeys.push_front(key);
        cache[key] = {chunk, lruKeys.begin()};
        cout << "[Cache] Stored: " << key << endl;
    }
};

// 3. Streaming Service
class StreamingService {
    CacheService cache;
public:
    StreamingService() : cache(3) {} // Max 3 chunks for demo

    vector<uint8_t> streamChunk(Song& song, long start, long end) {
        string key = song.id + "_" + to_string(start) + "_" + to_string(end);
        vector<uint8_t> data;

        if (cache.getChunk(key, data)) {
            cout << "Served from cache: " << key << endl;
            return data;
        }

        // Simulate fetching from Disk/S3
        cout << "Fetching from Storage... " << key << endl;
        data = vector<uint8_t>(end - start, 0xFF); // Mock data
        
        cache.putChunk(key, data);
        return data;
    }
};

int main() {
    Song s1 = {"S1", 1024};
    StreamingService streamer;

    streamer.streamChunk(s1, 0, 100);
    streamer.streamChunk(s1, 100, 200);
    streamer.streamChunk(s1, 0, 100); // Hits Cache
    streamer.streamChunk(s1, 200, 300);
    streamer.streamChunk(s1, 300, 400); // Evicts S1_100_200

    return 0;
}
