#include <memory>
#include <algorithm>

class YouTubeChannel : public Channel {
private:
    vector<weak_ptr<Subscriber>> subscribers; // Weak references!
    string channelName;

    // Helper to remove destroyed observers
    void cleanup() {
        subscribers.erase(
            remove_if(subscribers.begin(), subscribers.end(),
                      [](const weak_ptr<Subscriber>& w) { return w.expired(); }),
            subscribers.end());
    }

public:
    YouTubeChannel(string name) : channelName(name) {}

    void subscribe(shared_ptr<Subscriber> s) override {
        cleanup();
        subscribers.push_back(s);
    }

    void unsubscribe(shared_ptr<Subscriber> s) override {
        subscribers.erase(
            remove_if(subscribers.begin(), subscribers.end(),
                      [&](const weak_ptr<Subscriber>& w) {
                          auto locked = w.lock();
                          return !locked || locked.get() == s.get();
                      }),
            subscribers.end());
    }

    void notifySubscribers(const string& videoTitle) override {
        cleanup();
        for (auto& w : subscribers) {
            if (auto s = w.lock()) { // Promote to shared_ptr temporarily
                s->update(videoTitle); // Fire the event!
            }
        }
    }

    // The actual business logic
    void uploadVideo(string title) {
        cout << "\n[" << channelName << " uploaded: " << title << "]\n";
        notifySubscribers(title);
    }
};
