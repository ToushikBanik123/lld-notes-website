#include <thread>
#include <mutex>
#include <condition_variable>
#include <queue>

using namespace std;

queue<int> buffer;
const int MAX_SIZE = 5;

mutex mtx;
condition_variable cv_producer;
condition_variable cv_consumer;

void producer() {
    for (int i = 1; i <= 10; ++i) {
        unique_lock<mutex> lock(mtx);
        
        // Wait UNTIL buffer is not full. 
        // Automatically releases the lock while asleep!
        cv_producer.wait(lock, []() { return buffer.size() < MAX_SIZE; });
        
        buffer.push(i);
        cout << "Produced: " << i << "\n";
        
        // Wake up a sleeping consumer
        cv_consumer.notify_one();
    }
}

void consumer() {
    for (int i = 1; i <= 10; ++i) {
        unique_lock<mutex> lock(mtx);
        
        // Wait UNTIL buffer is not empty.
        cv_consumer.wait(lock, []() { return !buffer.empty(); });
        
        int item = buffer.front();
        buffer.pop();
        cout << "Consumed: " << item << "\n";
        
        // Wake up a sleeping producer
        cv_producer.notify_one();
    }
}

int main() {
    thread t1(producer);
    thread t2(consumer);
    
    t1.join();
    t2.join();
    
    return 0;
}
