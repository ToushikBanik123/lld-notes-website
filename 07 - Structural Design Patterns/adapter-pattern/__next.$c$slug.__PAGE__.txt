16:T427,// 1. Target Interface
class PaymentGateway {
public:
    virtual ~PaymentGateway() = default;
    virtual void pay(const string& orderId, double amount) = 0;
};

// 2. The Adaptee (Incompatible)
class RazorpayAPI {
public:
    void makePayment(const string& invoiceId, double amountInRupees) {
        cout << "Paid Rs. " << amountInRupees << " via Razorpay for invoice: " << invoiceId << "\n";
    }
};

// 3. The Adapter
class RazorpayAdapter : public PaymentGateway {
private:
    shared_ptr<RazorpayAPI> razorpayAPI;

public:
    RazorpayAdapter() {
        razorpayAPI = make_shared<RazorpayAPI>();
    }

    // Translates the expected interface to the Adaptee's interface
    void pay(const string& orderId, double amount) override {
        // We map 'orderId' to 'invoiceId' under the hood
        razorpayAPI->makePayment(orderId, amount);
    }
};

// 4. Client Code
int main() {
    // Client treats it like a standard PaymentGateway
    shared_ptr<PaymentGateway> gateway = make_shared<RazorpayAdapter>();
    gateway->pay("ORDER_999", 1500.00); 
}
