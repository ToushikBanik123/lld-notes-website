using namespace std;

// 1. Enums
enum class Direction { NORTH, EAST, SOUTH, WEST };

// Forward Declaration
class TrafficLight;

// 2. State Pattern
class TrafficLightState {
public:
    virtual void turnGreen(TrafficLight* light) = 0;
    virtual void turnYellow(TrafficLight* light) = 0;
    virtual void turnRed(TrafficLight* light) = 0;
    virtual string getName() const = 0;
    virtual ~TrafficLightState() = default;
};

// Traffic Light Context
class TrafficLight {
public:
    Direction dir;
    unique_ptr<TrafficLightState> state;

    TrafficLight(Direction d, unique_ptr<TrafficLightState> initialState) 
        : dir(d), state(move(initialState)) {}

    void setState(unique_ptr<TrafficLightState> newState) {
        state = move(newState);
    }

    void turnGreen() { state->turnGreen(this); }
    void turnYellow() { state->turnYellow(this); }
    void turnRed() { state->turnRed(this); }
};

// Concrete States
class RedState : public TrafficLightState {
public:
    void turnGreen(TrafficLight* light) override;
    void turnYellow(TrafficLight* light) override { throw runtime_error("Invalid Transition: RED to YELLOW"); }
    void turnRed(TrafficLight* light) override { /* Already Red */ }
    string getName() const override { return "RED"; }
};

class GreenState : public TrafficLightState {
public:
    void turnGreen(TrafficLight* light) override { /* Already Green */ }
    void turnYellow(TrafficLight* light) override;
    void turnRed(TrafficLight* light) override { throw runtime_error("Invalid Transition: GREEN to RED. Must pass through YELLOW."); }
    string getName() const override { return "GREEN"; }
};

class YellowState : public TrafficLightState {
public:
    void turnGreen(TrafficLight* light) override { throw runtime_error("Invalid Transition: YELLOW to GREEN"); }
    void turnYellow(TrafficLight* light) override { /* Already Yellow */ }
    void turnRed(TrafficLight* light) override;
    string getName() const override { return "YELLOW"; }
};

// Implement Transitions
void RedState::turnGreen(TrafficLight* light) {
    cout << "Turning Green" << endl;
    light->setState(make_unique<GreenState>());
}
void GreenState::turnYellow(TrafficLight* light) {
    cout << "Turning Yellow" << endl;
    light->setState(make_unique<YellowState>());
}
void YellowState::turnRed(TrafficLight* light) {
    cout << "Turning Red" << endl;
    light->setState(make_unique<RedState>());
}

// 3. The Orchestrator
class Intersection {
private:
    map<Direction, unique_ptr<TrafficLight>> lights;
    vector<Direction> cycleOrder = {Direction::NORTH, Direction::EAST, Direction::SOUTH, Direction::WEST};
    int currentPhaseIndex = 0;
    bool isEmergency = false;

public:
    Intersection() {
        lights[Direction::NORTH] = make_unique<TrafficLight>(Direction::NORTH, make_unique<RedState>());
        lights[Direction::EAST] = make_unique<TrafficLight>(Direction::EAST, make_unique<RedState>());
        lights[Direction::SOUTH] = make_unique<TrafficLight>(Direction::SOUTH, make_unique<RedState>());
        lights[Direction::WEST] = make_unique<TrafficLight>(Direction::WEST, make_unique<RedState>());
    }

    void handleEmergency(Direction emergencyDir) {
        cout << "\n🚨 EMERGENCY DETECTED AT DIRECTION " << (int)emergencyDir << " 🚨" << endl;
        isEmergency = true;
        
        // Safely transition all to RED
        for (auto& pair : lights) {
            auto& light = pair.second;
            if (light->state->getName() == "GREEN") light->turnYellow();
            if (light->state->getName() == "YELLOW") light->turnRed();
        }

        // Give green to emergency
        lights[emergencyDir]->turnGreen();
    }

    void cycleNext() {
        if (isEmergency) return;

        Direction currentDir = cycleOrder[currentPhaseIndex];
        Direction nextDir = cycleOrder[(currentPhaseIndex + 1) % 4];

        // Safely end current
        lights[currentDir]->turnYellow();
        lights[currentDir]->turnRed();

        // Safely start next
        lights[nextDir]->turnGreen();

        currentPhaseIndex = (currentPhaseIndex + 1) % 4;
    }
};

// Usage
int main() {
    Intersection intersection;

    intersection.lights[Direction::NORTH]->turnGreen(); // Start cycle manually for demo
    intersection.cycleNext(); // Turns North to Yellow->Red, East to Green

    intersection.handleEmergency(Direction::SOUTH); // Interrupts cycle

    return 0;
}
